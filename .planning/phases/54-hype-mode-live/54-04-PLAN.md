<!-- From 54-UI-SPEC. The pure function is the testable core. -->
export type IndicatorLedTier = "glow-faint" | "glow-soft" | "glow-strong" | "fault";
export interface IndicatorState {
  visible: boolean;          // false when mode !== "hype"
  label: string;             // "HYPE · LIVE" | "HYPE · LISTENING" | "HYPE · OFFLINE"
  ledTier: IndicatorLedTier; // amber tier or fault
  pulse: boolean;            // true within PULSE_WINDOW_MS of the latest reaction arrival
}
deriveIndicatorState({ mode, live, lastReactionAtMs, nowMs, reducedMotion }) -> IndicatorState
  - mode !== "hype" -> { visible:false, label:"", ledTier:"glow-faint", pulse:false }
  - !live -> { visible:true, label:"HYPE · OFFLINE", ledTier:"fault", pulse:false }
  - live, recent reaction (nowMs - lastReactionAtMs <= PULSE_WINDOW_MS) ->
      { visible:true, label:"HYPE · LIVE", ledTier: reducedMotion ? "glow-soft" : "glow-strong", pulse: !reducedMotion }
  - live, reaction within the "active" recency (e.g. <= ACTIVE_WINDOW_MS) -> "HYPE · LIVE", glow-soft, pulse false
  - live, idle (no recent reaction) -> "HYPE · LISTENING", glow-faint, pulse false
PULSE_WINDOW_MS ~600 (matches the UI-SPEC ramp); ACTIVE_WINDOW_MS ~10000 (a reaction in the last ~10s
  means "actively reacting" → LIVE; older → LISTENING). Both are module constants, one-line editable.
</indicator_state_design>
