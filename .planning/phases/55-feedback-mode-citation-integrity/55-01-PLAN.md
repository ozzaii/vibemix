CitationLinter (src/vibemix/coach/citation_linter.py):
  CitationLinter().check(text: str, registry_snapshot: dict | None, *, mode: str="live") -> LintResult
  LintResult fields: valid (bool), citations_found (int), missing (tuple[(src,body),...]), reason (str).
  reason ladder: "no_citations" → "malformed_atom" → "invalid_atoms" → "valid".
  Live tol = LIVE_TOLERANCE_S = 1.0; debrief tol = DEBRIEF_TOLERANCE_S = 2.0.

  reg.write(source, key, t_session); reg.snapshot() -> {src:{key:(t,...)}}; cold reg.snapshot()=={}.
  reg.write("ev","DROP",45.2) then snapshot() == {"ev":{"DROP":(45.2,)}}.

AICoach (src/vibemix/state/coach.py):
  AICoach.build_prompt(ev: Event, *, registry_snapshot=None, diet=False) -> str
  Non-diet returns f"[{evidence_line(...)} | event={ev.type}] {task_for_event(ev)}".
  evidence_line appends "evidence_corpus[ev=N,aud=M,mix=K]" ONLY when registry_snapshot is non-empty AND (ev_n+aud_n+mix_n)>0.
  evidence_line has NO phase= field (anti-hallucination invariant — do not assert its presence).
  task_for_event tails: PHASE → "Phase shifted: ...", MIX_MOVE → "...describe the SONIC EFFECT...", HEARTBEAT → "Steady stretch. ONE sharp observation...".

matrix (src/vibemix/prompts/matrix.py):
  build_system_instruction(skill: str, mode: str, ...) -> str  (skill in beginner/intermediate/pro; mode in hype/coach).
  COACH_BEGINNER / COACH_INTERMEDIATE / COACH_PRO each carry {mood_persona} + _ANTI_SLOP_FOOTER + (via build_system_instruction default) CITATION_GRAMMAR_BLOCK.
  CITATION_GRAMMAR_BLOCK contains the marker "--- CITATION GRAMMAR" and the forms "[ev:" / "[track:".
