```python
def _packaged_bank_dir() -> Path:
    """Return path to the bundled CC-BY exemplar bank.

    Resolves via importlib.resources so both dev (src/vibemix/) and wheel
    (PyInstaller / pip-installed) layouts work.
    """
    try:
        from importlib.resources import files
        return Path(str(files("vibemix.learn.assets.band_exemplars")))
    except Exception:
        return Path(__file__).parent / "assets" / "band_exemplars"


def _fallback_for_band(band: str) -> tuple[str, str, str] | None:
    """Return (synthetic_track_id, file_path, honest-null reason) or None.

    Sorts files alphabetically inside the band's subdir (deterministic);
    returns the first match. Synthetic track_id format is
    `_packaged:<band>:<file_stem>` so it never collides with library
    track ids.
    """
    bank_dir = _packaged_bank_dir() / band
    if not bank_dir.exists():
        return None
    files = sorted(bank_dir.glob("*.mp3")) + sorted(bank_dir.glob("*.ogg"))
    if not files:
        return None
    path = files[0]
    synthetic_id = f"_packaged:{band}:{path.stem}"
    return (synthetic_id, str(path), _HONEST_NULL_REASON)
```

```python
@dataclass
class ExemplarPick:
    track_id: str
    file_path: str
    band_score: float
    reason: str


class ExemplarFinder:
    """Picks band-exemplar tracks for EQ lessons. Pure-compute over the
    side-car band_shares table; falls back to the packaged CC-BY bank
    when library has < _LIBRARY_FLOOR tracks passing the floor.

    Invariant #2 binding. `registry.write("exemplar", track_id, t_session)`
    runs BEFORE find() returns so the AI's later [exemplar:<track_id>]
    citation resolves via EvidenceRegistry.has() during linter validation.
    A fabricated [exemplar:bogus] strips the whole turn because no
    `("exemplar", "bogus", _)` write ever happened.

    Note: as of Plan 93-04 the EVIDENCE_SOURCES frozenset still has 9
    sources (Plan 93-05 lands "exemplar" via the 4-site mirror); the
    registry's v1.0 permissive contract makes this write succeed today.
    """

    def __init__(self, *, registry: "EvidenceRegistry | None" = None) -> None:
        self._registry = registry
        self._library = None

    def _load_library(self):
        if self._library is not None:
            return self._library
        try:
            from vibemix.library.rekordbox import RekordboxLibrary
            self._library = RekordboxLibrary.try_load_cache()
        except Exception:
            return None
        return self._library

    def _resolve_library_path(self, lib, track_id):
        if lib is None:
            return None
        entry = lib.tracks.get(track_id) if hasattr(lib, "tracks") else None
        if entry is None:
            return None
        return entry.path if hasattr(entry, "path") else None

    def find(self, band: str, k: int = 1,
             t_session: float | None = None) -> list[ExemplarPick]:
        if t_session is None:
            t_session = time.time()

        from vibemix.learn.band_share_store import open_default_db, top_for_band
        try:
            with open_default_db() as conn:
                rows = top_for_band(conn, band, k=max(k, _LIBRARY_FLOOR),
                                    max_kick_corr=_KICK_GUARD_R)
        except Exception:
            rows = []

        picks: list[ExemplarPick] = []
        if len(rows) >= _LIBRARY_FLOOR:
            lib = self._load_library()
            for track_id, score, _kick in rows[:k]:
                file_path = self._resolve_library_path(lib, track_id)
                if file_path is None:
                    continue
                if self._registry is not None:
                    self._registry.write("exemplar", track_id, t_session)
                picks.append(ExemplarPick(
                    track_id=track_id,
                    file_path=file_path,
                    band_score=score,
                    reason=f"from your library — strongest {band}-band track",
                ))

        if picks:
            return picks

        fb = _fallback_for_band(band)
        if fb is None:
            return []
        synthetic_id, file_path, reason = fb
        if self._registry is not None:
            self._registry.write("exemplar", synthetic_id, t_session)
        return [ExemplarPick(
            track_id=synthetic_id,
            file_path=file_path,
            band_score=0.0,
            reason=reason,
        )]
```
