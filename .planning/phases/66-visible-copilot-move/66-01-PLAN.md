```python
def _build_citation_strip(*, reaction_text: str, registry: EvidenceRegistry) -> list[dict]:
    snapshot = registry.snapshot()
    chips: list[dict] = []
    for source, body in parse_citations(reaction_text):
        if len(chips) >= CITATION_STRIP_MAX_CHIPS:
            break
        # CURRENT (Phase 65 deliberately deferred): no 'recall' in this tuple
        if source not in ("ev", "mix", "midi", "key"):
            continue
        inner = snapshot.get(source, {})
        timestamps = inner.get(body)
        if not timestamps:
            continue
        timestamp_s = float(timestamps[0])
        if source == "key":
            verb = "key"   # fixed letters-only verb (Phase 59 / DECK-03 precedent)
        else:
            key, _, _ = body.partition("@")
            verb_tokens = key.lower().split("_")
            verb = " ".join(verb_tokens[:CITATION_VERB_MAX_WORDS])
        chips.append({"event_id": f"{source}:{body}", "verb": verb, "timestamp_s": timestamp_s})
    return chips
```
