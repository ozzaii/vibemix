    def _walk_string_values(node: Any) -> Iterator[str]:
        """Yield every string descendant of ``node`` — depth-first.
        For dicts: yield values; recurse into nested dict / list values.
        For lists: recurse into elements.
        For strings: yield the string.
        Skip booleans / ints / floats / None.

    def check_no_tutor_slop(
        transcripts_dir: Path,
        *,
        quiet: bool = False,
    ) -> int:
        """Walk ``transcripts_dir`` for `**/*.json`. For each fixture:
        json.loads → walk all string descendants → scan for blocklist tokens.
        Return 0 if no fixture trips the blocklist; return 1 if any does.
        On gate-fail, stderr prints one line per offending (file, token, category).
        """

    def main(argv: list[str] | None = None) -> int:
