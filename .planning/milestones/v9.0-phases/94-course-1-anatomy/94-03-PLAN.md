    ```python
    class ExemplarLessonController:
        """Drives the L1.14 EQ-as-Tutor 3-band cycle.

        Reads exemplar_cycle from the fixture script. For each band (low, mid,
        high in fixture order):
          1. Calls ExemplarFinder.find(band, k=1) — pre-registers the pick in
             EvidenceRegistry as Invariant #2 binding.
          2. Emits LearnExemplarPlay envelope to the shell.
          3. Calls ExemplarPlayer.play(file_path) IF can_play() and file_path
             is non-empty.
          4. Emits LearnTutorSpeak with the fixture's per-band copy + a
             citations[] containing the `[exemplar:<track_id>]` atom.

        On user MIDI matching the current band's expected_action: emits
        LearnExemplarStop + LearnAdvance, advances cycle index. After all
        3 bands: emits LearnCompleteLesson and stops the player.

        Honest-null path: if ExemplarFinder returns [], skip step 3 (no audio),
        emit LearnTutorSpeak with a controller-baked tutor line that names
        the honest-null reason ("your library doesn't have a great example
        of <band>; we'd usually play a packaged track but the bank is also
        empty — turn the knob anyway to feel the change."), and STILL gate
        cycle advance on the user's MIDI. Lesson completes after all 3 bands
        regardless. Cite no exemplar in this branch.
        """

        def __init__(
            self,
            *,
            finder: ExemplarFinder,
            player: ExemplarPlayer,
            ipc_emit: Callable[[dict[str, Any]], None],
        ) -> None:
            self._finder = finder
            self._player = player
            self._emit = ipc_emit
            self._cycle: list[dict[str, Any]] = []
            self._active_idx: int = -1
            self._active_pick: ExemplarPick | None = None

        def start(self, *, script: dict[str, Any], lesson_id: str = "L1.14") -> None:
            """Begin the 3-band cycle from a lesson script (already loaded).
            ``script`` is the JSON dict from CURRICULUM["L1.14"].script.
            """
            cycle = script.get("exemplar_cycle")
            if not isinstance(cycle, list) or not cycle:
                # Defensive: fixture missing exemplar_cycle — emit
                # complete_lesson immediately (degraded) and bail.
                self._emit(
                    LearnCompleteLesson.make(
                        lesson_id=lesson_id, reason="exemplar_cycle_missing"
                    ).to_dict()
                )
                return
            self._cycle = list(cycle)
            self._active_idx = -1
            self._advance(lesson_id=lesson_id)

        def matches(self, midi: dict[str, Any]) -> bool:
            """Return True iff midi matches the active band's expected_action.
            Reuses the same logic shape as LessonRuntime.action_matches
            (runtime.py:209-300) — CC delta branch + button branch.
            Caller (LessonRuntime hook) calls this to decide whether to
            invoke .ack() vs let LessonRuntime's default gate run.
            """
            if self._active_idx < 0 or self._active_idx >= len(self._cycle):
                return False
            expected = self._cycle[self._active_idx].get("expected_action")
            if not isinstance(expected, dict):
                return False
            return _midi_matches(midi, expected)

        def ack(self, *, lesson_id: str = "L1.14") -> None:
            """Advance to the next band (or finish if done)."""
            self._stop_active(lesson_id=lesson_id)
            self._advance(lesson_id=lesson_id)

        def stop(self, *, lesson_id: str = "L1.14") -> None:
            """Tear down — stop player + emit exemplar_stop + reset state.
            Idempotent: safe to call from any state.
            """
            self._stop_active(lesson_id=lesson_id)
            self._active_idx = -1
            self._cycle = []

        # --- internal helpers --------------------------------------------

        def _advance(self, *, lesson_id: str) -> None:
            self._active_idx += 1
            if self._active_idx >= len(self._cycle):
                # All 3 bands done — emit complete_lesson and bail.
                self._emit(
                    LearnCompleteLesson.make(
                        lesson_id=lesson_id, reason="exemplar_cycle_complete"
                    ).to_dict()
                )
                return

            entry = self._cycle[self._active_idx]
            band = entry.get("band", "")
            picks = self._finder.find(band, k=1, t_session=time.time())
            if not picks:
                # Honest-null fallback — no audio, but still emit tutor copy.
                self._active_pick = None
                self._emit_honest_null_speak(band=band, entry=entry, lesson_id=lesson_id)
                return

            pick = picks[0]
            self._active_pick = pick
            # Emit exemplar_play (shell paints "now playing" chip)
            self._emit(
                LearnExemplarPlay.make(
                    track_id=pick.track_id,
                    band=band,
                    file_path=pick.file_path,
                ).to_dict()
            )
            # Start audio (idempotent — player handles can_play guard internally)
            if pick.file_path:
                try:
                    self._player.play(pick.file_path)
                except Exception as exc:  # pragma: no cover — defensive
                    import sys
                    print(f"[learn.exemplar_lesson] play failed: {exc!r}", file=sys.stderr)
            # Emit tutor speak with the citation atom
            tutor_text = str(entry.get("tutor_speak", "")).strip()
            self._emit(
                LearnTutorSpeak.make(
                    lesson_id=lesson_id,
                    beat=self._active_idx,
                    text=tutor_text,
                    tts_marker=f"L114.band_{band}",
                    citations=[{"source": "exemplar", "key": pick.track_id}],
                ).to_dict()
            )

        def _emit_honest_null_speak(
            self, *, band: str, entry: dict[str, Any], lesson_id: str
        ) -> None:
            """Tutor copy for the degraded-install branch — neither library
            nor packaged bank had a track. No citation (no exemplar to cite)."""
            # Hand-bake the honest-null phrasing; mirrors the constant in
            # exemplar.py:_HONEST_NULL_REASON but reshaped as a tutor beat.
            fallback_text = (
                f"i don't have an example of the {band} band ready right now. "
                f"turn deck a's {band} eq knob anyway — you'll still feel the change."
            )
            self._emit(
                LearnTutorSpeak.make(
                    lesson_id=lesson_id,
                    beat=self._active_idx,
                    text=fallback_text,
                    tts_marker=f"L114.band_{band}_honest_null",
                    citations=[],
                ).to_dict()
            )

        def _stop_active(self, *, lesson_id: str) -> None:
            if self._active_pick is not None:
                self._emit(
                    LearnExemplarStop.make(
                        track_id=self._active_pick.track_id,
                        lesson_id=lesson_id,
                    ).to_dict()
                )
            try:
                self._player.stop()
            except Exception as exc:  # pragma: no cover — defensive
                import sys
                print(f"[learn.exemplar_lesson] stop failed: {exc!r}", file=sys.stderr)
            self._active_pick = None
    ```

    _RECITAL_SUBSET_SIZE: int = 5  # CURR-1.16 lock — 5 prompts per recital


    class RecitalRuntime:
        """L1.16 Course 1 Recital — 5-prompt mixed gate.

        On .start(script, lesson_id), samples ``_RECITAL_SUBSET_SIZE`` entries
        from ``script["recital_pool"]`` using a daily-rotating seed (so a
        replay within one day re-tests the same 5; a replay tomorrow re-rolls).

        For each prompt:
          - emit LearnTutorSpeak with the prompt text
          - emit LearnHighlight if the prompt's expected_action carries a
            controllable region (CC drop on a knob/fader/jog OR button press
            on a transport key)
          - on matching MIDI: score +1, advance to next prompt
          - on non-matching MIDI: stay on current prompt (no penalty; recital
            is honest grading — performed-correctly only)

        On full cycle:
          - score == 5: set LearnProgress.course_2_unlocked = True +
            save_progress; emit LearnTutorSpeak with the pass copy; emit
            LearnProgressState (so the UI re-paints the unlock).
          - score < 5: course_2_unlocked stays False; emit fail copy with
            {score} substituted; lesson is complete (the user can re-roll
            tomorrow OR replay-now and re-attempt — replay-now reshuffles
            because seed changes only at day boundary, BUT replay-now within
            the SAME day picks the SAME subset).
          - either way: emit LearnCompleteLesson with reason="recital_pass" or
            "recital_fail".

        Why same subset on same-day replay? Per CONTEXT.md §decisions §recital:
        "honest grading" — the user must perform the SAME 5 correctly; rolling
        a different 5 on each retry would let them grind a favorable
        combination. Day-rolling means a frustrated user can sleep on it and
        get a fresh combination tomorrow — anti-frustration without
        retry-grinding.
        """

        def __init__(
            self,
            *,
            ipc_emit: Callable[[dict[str, Any]], None],
            progress: LearnProgress,
            seed: int | None = None,
            save_fn: Callable[[LearnProgress], Any] = save_progress,
            rng_class: type = random.Random,
        ) -> None:
            self._emit = ipc_emit
            self._progress = progress
            self._seed = seed
            self._save = save_fn
            self._rng_class = rng_class
            self._sampled: list[dict[str, Any]] = []
            self._active_idx: int = -1
            self._score: int = 0
            self._outcomes: dict[str, str] = {}

        def start(self, *, script: dict[str, Any], lesson_id: str = "L1.16") -> None:
            pool = script.get("recital_pool")
            self._outcomes = script.get("recital_outcomes") or {}
            if not isinstance(pool, list) or len(pool) < _RECITAL_SUBSET_SIZE:
                # Degraded fixture — emit complete_lesson immediately.
                self._emit(
                    LearnCompleteLesson.make(
                        lesson_id=lesson_id, reason="recital_pool_too_small"
                    ).to_dict()
                )
                return
            seed = self._seed if self._seed is not None else int(time.time() // 86400)
            rng = self._rng_class(seed)
            self._sampled = rng.sample(pool, k=_RECITAL_SUBSET_SIZE)
            self._active_idx = -1
            self._score = 0
            self._advance(lesson_id=lesson_id)

        def matches(self, midi: dict[str, Any]) -> bool:
            if self._active_idx < 0 or self._active_idx >= len(self._sampled):
                return False
            expected = self._sampled[self._active_idx].get("expected_action")
            if not isinstance(expected, dict):
                return False
            return _midi_matches(midi, expected)

        def ack(self, *, lesson_id: str = "L1.16") -> None:
            self._score += 1
            self._advance(lesson_id=lesson_id)

        def stop(self, *, lesson_id: str = "L1.16") -> None:
            self._sampled = []
            self._active_idx = -1
            self._score = 0

        # --- internal -----------------------------------------------------

        def _advance(self, *, lesson_id: str) -> None:
            self._active_idx += 1
            if self._active_idx >= len(self._sampled):
                self._finalize(lesson_id=lesson_id)
                return
            entry = self._sampled[self._active_idx]
            # Emit prompt as tutor_speak
            self._emit(
                LearnTutorSpeak.make(
                    lesson_id=lesson_id,
                    beat=self._active_idx,
                    text=str(entry.get("prompt", "")).strip(),
                    tts_marker=f"L116.recital{self._active_idx + 1}",
                    citations=[],
                ).to_dict()
            )
            # Emit highlight if there's a renderable target (e.g. control name).
            expected = entry.get("expected_action") or {}
            control = expected.get("control")
            if control:
                try:
                    self._emit(
                        LearnHighlight.make(
                            lesson_id=lesson_id,
                            control_id=str(control),
                            deck=expected.get("deck"),
                            kind="pulse",
                        ).to_dict()
                    )
                except Exception as exc:  # pragma: no cover — defensive
                    print(f"[learn.recital] highlight emit failed: {exc!r}", file=sys.stderr)

        def _finalize(self, *, lesson_id: str) -> None:
            passed = self._score >= _RECITAL_SUBSET_SIZE
            if passed:
                self._progress.course_2_unlocked = True
                try:
                    self._save(self._progress)
                except Exception as exc:  # pragma: no cover — defensive
                    print(f"[learn.recital] save_progress failed: {exc!r}", file=sys.stderr)
                copy = self._outcomes.get("pass", "5 of 5. course 2 unlocked.")
                self._emit(
                    LearnTutorSpeak.make(
                        lesson_id=lesson_id,
                        beat=self._active_idx,
                        text=copy,
                        tts_marker="L116.outcome_pass",
                        citations=[],
                    ).to_dict()
                )
                # Repaint the lesson dots in the HUD + signal unlock to the shell.
                try:
                    self._emit(
                        LearnProgressState.make(
                            snapshot=self._progress.snapshot()
                        ).to_dict()
                    )
                except Exception as exc:  # pragma: no cover — defensive
                    print(f"[learn.recital] progress_state emit failed: {exc!r}", file=sys.stderr)
                self._emit(
                    LearnCompleteLesson.make(
                        lesson_id=lesson_id, reason="recital_pass"
                    ).to_dict()
                )
            else:
                copy_template = self._outcomes.get("fail", "{score} of 5. replay when you're ready.")
                copy = copy_template.replace("{score}", str(self._score))
                self._emit(
                    LearnTutorSpeak.make(
                        lesson_id=lesson_id,
                        beat=self._active_idx,
                        text=copy,
                        tts_marker="L116.outcome_fail",
                        citations=[],
                    ).to_dict()
                )
                self._emit(
                    LearnCompleteLesson.make(
                        lesson_id=lesson_id, reason="recital_fail"
                    ).to_dict()
                )
